## 概要

AEO戦略資料のデザイン要素をGEOスコアリングダッシュボードに適用し、
「データドリブンな信頼感」と「視覚的インパクト」を両立させる改善案です。

| 項目 | 現状 (v2.1) | 改善後 (v2.1.1) |

### 2.2 新しいフォーマット（コード例付き）

```python
def render_before_after_kpi(
    before: float,
    after: float,
    label: str,
    unit: str = "pt",
    max_value: float = 100
) -> str:
    """Before/After形式のKPIカードを生成

    Args:
        before: 変更前の値
        after: 変更後の値（現在値）
        label: ラベル名
        unit: 単位（pt, %, 件 など）
        max_value: 最大値（プログレスバー用）

    Returns:
        HTML文字列
    """
    delta = after - before
    delta_sign = "+" if delta >= 0 else ""
    delta_color = COLORS_AEO["accent_primary"] if delta >= 0 else COLORS_AEO["error"]
    progress_pct = (after / max_value) * 100

    return f'''
    <div class="kpi-card-aeo">
        <div class="kpi-label-aeo">{label}</div>
        <div class="kpi-before-after">
            <span class="kpi-before">{before:.0f}</span>
            <span class="kpi-arrow">→</span>
            <span class="kpi-after">{after:.0f}</span>
        </div>
        <div class="kpi-delta-aeo" style="color: {delta_color};">
            {delta_sign}{delta:.0f}{unit}
        </div>
        <div class="kpi-progress-bar">
            <div class="kpi-progress-fill" style="width: {progress_pct}%;"></div>
        </div>
    </div>
    '''
```

### 3.2 Plotlyコード例

```python
def render_engine_scores(scores: dict) -> go.Figure:
    """エンジン別スコアの横棒グラフを生成

    Args:
        scores: {
            "ChatGPT": {"current": 78, "previous": 66},
            "Claude": {"current": 65, "previous": 68},
            "Gemini": {"current": 72, "previous": 64},
            "Perplexity": {"current": 82, "previous": 67},
        }

    Returns:
        Plotly Figure
    """
    engines = list(scores.keys())
    current_scores = [scores[e]["current"] for e in engines]
    previous_scores = [scores[e]["previous"] for e in engines]
    deltas = [c - p for c, p in zip(current_scores, previous_scores)]

    # 色の決定（変化に応じて）
    colors = [
        COLORS_AEO["accent_primary"] if d >= 0 else COLORS_AEO["error"]
        for d in deltas
    ]

    fig = go.Figure()

    # 現在のスコア（メインバー）
    fig.add_trace(go.Bar(
        y=engines[::-1],
        x=current_scores[::-1],
        orientation='h',
        marker=dict(
            color=colors[::-1],
            line=dict(width=0),
            cornerradius=8
        ),
        text=[f"{s} ({'+' if d >= 0 else ''}{d})"
              for s, d in zip(current_scores[::-1], deltas[::-1])],
        textposition='inside',
        textfont=dict(
            color=COLORS_AEO["background"],
            size=14,
            family="JetBrains Mono"
        ),
        hovertemplate=(
            '<b>%{y}</b><br>'
            'Current: %{x}<br>'
            'Previous: %{customdata[0]}<br>'
            'Change: %{customdata[1]:+d}<extra></extra>'
        ),
        customdata=list(zip(previous_scores[::-1], deltas[::-1]))
    ))

    # 前回スコア（マーカー）
    fig.add_trace(go.Scatter(
        y=engines[::-1],
        x=previous_scores[::-1],
        mode='markers',
        marker=dict(
            symbol='line-ns-open',
            size=20,
            line=dict(width=2, color=COLORS_AEO["text_muted"]),
            color='rgba(0,0,0,0)'
        ),
        name='前回',
        hoverinfo='skip'
    ))

    fig.update_layout(
        paper_bgcolor='rgba(0,0,0,0)',
        plot_bgcolor='rgba(0,0,0,0)',
        font=dict(
            family="'Noto Sans JP', sans-serif",
            color=COLORS_AEO["text_secondary"]
        ),
        height=300,
        margin=dict(l=100, r=40, t=20, b=40),
        xaxis=dict(
            range=[0, 100],
            gridcolor='rgba(148, 163, 184, 0.1)',
            zeroline=False,
            tickfont=dict(size=12),
            title=dict(
                text='スコア',
                font=dict(size=12, color=COLORS_AEO["text_muted"])
            )
        ),
        yaxis=dict(
            tickfont=dict(size=14, color=COLORS_AEO["text_primary"]),
        ),
        showlegend=False,
        bargap=0.3
    )

    return fig
```

### 4.2 積み上げ棒グラフ vs 円グラフの比較

| 特性 | 積み上げ棒グラフ | 円グラフ |
|------|------------------|----------|
| 時系列比較 | ◎ 優れている | △ 難しい |
| 正確な比較 | ◎ 長さで比較可能 | △ 角度は不正確 |
| 多数のカテゴリ | ○ 対応可能 | × 5以上は困難 |
| 視覚的インパクト | ○ | ◎ |
| AEOサイトとの整合性 | ◎ 採用されている | △ 不採用 |

**推奨**: 積み上げ棒グラフ（時系列比較を重視）

### 4.3 Plotlyコード例

```python
def render_sov_chart(market_data: dict, category: str = "CRM") -> go.Figure:
    """SOV（Share of Voice）チャートを生成

    Args:
        market_data: {
            "自社": 32,
            "競合A": 18,
            "競合B": 15,
            "競合C": 12,
            "その他": 23
        }
        category: カテゴリ名

    Returns:
        Plotly Figure
    """
    brands = list(market_data.keys())
    shares = list(market_data.values())

    # 色の割り当て（自社のみアクセントカラー）
    colors = [
        COLORS_AEO["accent_primary"] if b == "自社" else
        f"rgba(148, 163, 184, {0.8 - i*0.15})"
        for i, b in enumerate(brands)
    ]

    fig = go.Figure()

    cumulative = 0
    for brand, share, color in zip(brands, shares, colors):
        fig.add_trace(go.Bar(
            name=brand,
            x=[share],
            y=[category],
            orientation='h',
            marker=dict(
                color=color,
                line=dict(width=0),
                cornerradius=0
            ),
            text=f"{brand}<br>{share}%",
            textposition='inside',
            textfont=dict(
                size=12 if share >= 10 else 10,
                color=COLORS_AEO["background"] if brand == "自社"
                      else COLORS_AEO["text_primary"],
                family="'Noto Sans JP', sans-serif"
            ),
            hovertemplate=f'<b>{brand}</b><br>シェア: {share}%<extra></extra>',
            base=cumulative
        ))
        cumulative += share

    fig.update_layout(
        barmode='stack',
        paper_bgcolor='rgba(0,0,0,0)',
        plot_bgcolor='rgba(0,0,0,0)',
        height=120,
        margin=dict(l=20, r=20, t=20, b=20),
        xaxis=dict(
            range=[0, 100],
            showgrid=False,
            showticklabels=False,
            zeroline=False
        ),
        yaxis=dict(
            showticklabels=False
        ),
        showlegend=False,
        bargap=0
    )

    return fig
```

### 5.2 段階的進捗の可視化

```python
def render_progress_timeline(months_data: list) -> str:
    """段階的進捗タイムラインを生成

    Args:
        months_data: [
            {
                "phase": "Phase 1",
                "title": "基盤構築",
                "start_month": 1,
                "end_month": 3,
                "tasks": [
                    {"name": "技術SEO監査", "start": 1, "end": 2, "status": "completed"},
                    {"name": "コンテンツ最適化", "start": 2, "end": 4, "status": "in_progress"},
                    {"name": "E-E-A-T強化", "start": 3, "end": 5, "status": "pending"},
                ]
            },
            ...
        ]

    Returns:
        HTML文字列
    """
    html = '''
    <div class="timeline-container-aeo">
    '''

    for phase_data in months_data:
        html += f'''
        <div class="timeline-phase">
            <div class="phase-header">
                <span class="phase-badge">{phase_data["phase"]}</span>
                <span class="phase-title">{phase_data["title"]}</span>
            </div>
            <div class="month-grid">
                {''.join([f'<div class="month-cell">M{i}</div>' for i in range(1, 13)])}
            </div>
            <div class="timeline-bars">
        '''

        for task in phase_data["tasks"]:
            status_class = f"status-{task['status']}"
            left_pct = (task["start"] - 1) / 12 * 100
            width_pct = (task["end"] - task["start"] + 1) / 12 * 100

            html += f'''
                <div class="task-row">
                    <div class="task-name">{task["name"]}</div>
                    <div class="task-bar-container">
                        <div class="task-bar {status_class}"
                             style="left: {left_pct}%; width: {width_pct}%;">
                        </div>
                    </div>
                </div>
            '''

        html += '''
            </div>
        </div>
        '''

    html += '</div>'
    return html
```
